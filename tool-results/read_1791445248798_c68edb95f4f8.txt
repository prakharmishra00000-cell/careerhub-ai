    61→function orDash(v: any): string {
    62→  if (v === null || v === undefined || v === '') return 'Not specified'
    63→  if (typeof v === 'number' && v === 0) return 'Not specified'
    64→  return String(v)

    67→function CompletionRing({ value, size = 96 }: { value: number; size?: number }) {
    68→  const stroke = 8
    69→  const r = (size - stroke) / 2
    70→  const c = 2 * Math.PI * r
    71→  const offset = c - (value / 100) * c
    72→  return (
    73→    <svg width={size} height={size} className="shrink-0 -rotate-90">
    74→      <circle cx={size / 2} cy={size / 2} r={r} stroke="currentColor" strokeWidth={stroke} className="text-muted/40" fill="none" />
    75→      <circle
    76→        cx={size / 2} cy={size / 2} r={r}
    77→        stroke="currentColor" strokeWidth={stroke} fill="none"
    78→        className="text-primary transition-all duration-500"
    79→        strokeDasharray={c}
    80→        strokeDashoffset={offset}
    81→        strokeLinecap="round"
    82→      />
    83→    </svg>
    84→  )
    85→}

   565→function SelectInput({ label, value, onChange, options }: {

   567→  value: string
   568→  onChange: (v: string) => void

   580→          ))}
   581→        </SelectContent>
   582→      </Select>
   583→    </div>
   584→  )
